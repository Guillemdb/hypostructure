\title[Scale Collapse for Type II Rescalings]{Scale-Collapse Cost Exclusion for Type II Navier--Stokes Rescalings}
\author{Guillem Cordero}
\date{}

\begin{document}

\begin{abstract}
We analyze Type II Navier--Stokes rescalings for which the renormalized scale
collapses along the tail.  In a scale-translation gauge, the logarithmic scale
satisfies \(\partial_\tau\log\lambda=a\).  If the collapsing rescaling has a
nonzero localized \(L^2\) core, then the negative part of the scale drift has
infinite localized accumulation.  This infinite accumulation is itself the
scale-collapse Type II cost.  Consequently finite-cost Type II branches cannot
realize genuine scale collapse with a nonvanishing localized core.  An absolute
scale-drift cost gives the same conclusion by pointwise domination.  We also
record the autonomous stationary-profile reduction as a secondary rigidity
route; it is not needed for the finite-cost scale-collapse exclusion.
This paper uses the standard Navier--Stokes theory for suitable weak solutions (Leray \cite{Leray1934}, Caffarelli--Kohn--Nirenberg \cite{CaffarelliKohnNirenberg1982}, Scheffer \cite{Scheffer1976}, and Lin \cite{Lin1998}).
\end{abstract}

\section{Renormalized scale collapse}

Let \((u,p)\) be a suitable weak solution of the three-dimensional incompressible
Navier--Stokes equations
\[
  \partial_tu+(u\cdot\nabla)u+\nabla p=\nu\Delta u,
  \qquad \nabla\cdot u=0,
\]
near a putative Type II singular time.  We consider a terminal rescaling written
in a scale-translation gauge:
\[
  u(x,t)=\lambda(\tau)^{-1}V\left(\frac{x-X(\tau)}{\lambda(\tau)},\tau\right),
  \qquad
  p(x,t)=\lambda(\tau)^{-2}P\left(\frac{x-X(\tau)}{\lambda(\tau)},\tau\right)+c(t),
\]
where \(\lambda(\tau)>0\).  The corresponding renormalized equation is
\begin{equation}\label{eq:renorm-ns}
  \partial_\tau V+(V\cdot\nabla)V+\nabla P
  =\nu\Delta V+a(\tau)(V+y\cdot\nabla V)+b(\tau)\cdot\nabla V,
  \qquad \nabla\cdot V=0.
\end{equation}
The scale convention is
\begin{equation}\label{eq:log-scale-convention}
  \frac{d}{d\tau}\log\lambda(\tau)=a(\tau).
\end{equation}
Thus physical scale collapse \(\lambda(\tau)\to0\) is associated with negative
logarithmic scale drift.

Let \(\Phi(\tau,\cdot)\in C_c^\infty(\mathbb R^3)\) be a nonnegative moving
cutoff adapted to the local scale-collapse core, and define
\[
  M_R(\tau)=\int_{\mathbb R^3}|V(y,\tau)|^2\Phi(\tau,y)\,dy.
\]
A nonvanishing localized core means that there are \(m_0>0\) and
\(\tau_1\ge\tau_0\) such that
\[
  M_R(\tau)\ge m_0
  \qquad\text{for a.e. }\tau\ge\tau_1.
\]

\begin{definition}[Scale-collapse and absolute scale costs]
The scale-collapse cost density is
\[
  \mathfrak C_{\rm sc}(\tau)=a_-(\tau)M_R(\tau),
  \qquad a_-(\tau)=\max\{-a(\tau),0\}.
\]
The absolute scale-drift cost density is
\[
  \mathfrak C_{\rm abs}(\tau)
  =\nu\int_{\mathbb R^3}|\nabla V(y,\tau)|^2\Phi(\tau,y)\,dy
  +|a(\tau)|M_R(\tau).
\]
The corresponding cumulative costs on \([\tau_0,\infty)\) are the integrals of
these nonnegative densities.
\end{definition}

\begin{definition}[Finite-cost scale-collapse branch]\label{def:finite-cost-branch}
A scale-collapse branch is finite-cost in the scale-collapse sense if
\[
  \int_{\tau_0}^\infty \mathfrak C_{\rm sc}(\tau)\,d\tau<\infty.
\]
It is finite-cost in the absolute scale-drift sense if
\[
  \int_{\tau_0}^\infty \mathfrak C_{\rm abs}(\tau)\,d\tau<\infty.
\]
\end{definition}

\section{Scale-drift divergence}

\begin{lemma}[Finite-or-infinite alternative]\label{lem:finite-infinite}
For every measurable renormalized solution for which \(a_-M_R\) is measurable
and nonnegative, exactly one of the following alternatives holds:
\[
  \int_{\tau_0}^\infty a_-(\tau)M_R(\tau)\,d\tau<\infty,
  \qquad\text{or}\qquad
  \int_{\tau_0}^\infty a_-(\tau)M_R(\tau)\,d\tau=\infty.
\]
\end{lemma}

\begin{proof}
The integral is an extended nonnegative number.  Hence it is either finite or
infinite, and the two cases are mutually exclusive.
\end{proof}

\begin{lemma}[Logarithmic scale identity]\label{lem:log-scale}
Under the convention \eqref{eq:log-scale-convention}, for every
\(\tau_0\le\tau_1<\tau_2<\infty\),
\[
  \int_{\tau_1}^{\tau_2}a(\tau)\,d\tau
  =\log\lambda(\tau_2)-\log\lambda(\tau_1).
\]
\end{lemma}

\begin{proof}
This is the fundamental theorem of calculus applied to
\(\partial_\tau\log\lambda=a\).
\end{proof}

\begin{theorem}[Collapse with a nonvanishing core forces infinite scale-collapse cost]\label{thm:collapse-core-floor}
If
\[
  \lambda(\tau)\to0\qquad\text{as }\tau\to\infty,
\]
and the rescaling has a nonvanishing localized core, then
\[
  \int_{\tau_0}^\infty a_-(\tau)M_R(\tau)\,d\tau=
  \int_{\tau_0}^\infty \mathfrak C_{\rm sc}(\tau)\,d\tau=
  \infty.
\]
\end{theorem}

\begin{proof}
By \cref{lem:log-scale},
\[
  \int_{\tau_1}^{\tau_2}a(\tau)\,d\tau
  =\log\lambda(\tau_2)-\log\lambda(\tau_1).
\]
Since \(\lambda(\tau)\to0\), the right-hand side tends to \(-\infty\) as
\(\tau_2\to\infty\).  If \(\int_{\tau_1}^\infty a_-(\tau)\,d\tau<\infty\), then
\[
  \int_{\tau_1}^{\tau_2}a(\tau)\,d\tau
  =\int_{\tau_1}^{\tau_2}a_+(\tau)\,d\tau
  -\int_{\tau_1}^{\tau_2}a_-(\tau)\,d\tau
  \ge -\int_{\tau_1}^\infty a_-(\tau)\,d\tau> -\infty,
\]
contradicting the divergence to \(-\infty\).  Thus
\(\int_{\tau_1}^\infty a_-(\tau)\,d\tau=\infty\).

The core floor gives \(M_R(\tau)\ge m_0\) for a.e. \(\tau\ge\tau_1\).  Hence
\[
  \int_{\tau_0}^\infty a_-(\tau)M_R(\tau)\,d\tau
  \ge m_0\int_{\tau_1}^\infty a_-(\tau)\,d\tau=
  \infty.
\]
\end{proof}

\section{Finite-cost exclusion}

\begin{theorem}[Scale-collapse cost exclusion]\label{thm:cost-exclusion}
No finite-cost scale-collapse branch in the sense of
\cref{def:finite-cost-branch} can satisfy both
\(\lambda(\tau)\to0\) and the nonvanishing localized-core condition.
\end{theorem}

\begin{proof}
By \cref{thm:collapse-core-floor}, genuine scale collapse with a nonvanishing
localized core gives
\[
  \int_{\tau_0}^\infty \mathfrak C_{\rm sc}(\tau)\,d\tau=\infty.
\]
This contradicts finite cost in the scale-collapse sense.
\end{proof}

\begin{lemma}[Absolute cost domination]\label{lem:absolute-cost}
For every \(\tau\),
\[
  \mathfrak C_{\rm abs}(\tau)\ge \mathfrak C_{\rm sc}(\tau).
\]
Consequently,
\[
  \int_{\tau_0}^\infty \mathfrak C_{\rm sc}(\tau)\,d\tau=\infty
  \quad\Longrightarrow\quad
  \int_{\tau_0}^\infty \mathfrak C_{\rm abs}(\tau)\,d\tau=\infty.
\]
\end{lemma}

\begin{proof}
Pointwise,
\[
  \mathfrak C_{\rm abs}(\tau)
  =\nu\int |\nabla V|^2\Phi+|a(\tau)|M_R(\tau)
  \ge |a(\tau)|M_R(\tau)
  \ge a_-(\tau)M_R(\tau)=\mathfrak C_{\rm sc}(\tau).
\]
Integrating gives the implication.
\end{proof}

\begin{corollary}[Absolute scale-cost exclusion]\label{cor:absolute-cost-exclusion}
No finite-cost scale-collapse branch in the absolute scale-drift sense can
satisfy both \(\lambda(\tau)\to0\) and the nonvanishing localized-core condition.
\end{corollary}

\begin{proof}
Combine \cref{thm:collapse-core-floor,lem:absolute-cost}.  The resulting
infinite absolute scale-drift cost contradicts finite cost in the absolute
scale-drift sense.
\end{proof}

\section{Autonomous rigidity route}

The cost route above is the scale-collapse exclusion used in the finite-cost
Type II analysis.  There is also a secondary PDE rigidity route.  It is useful
when one wants to identify the scale-collapse obstruction with a stationary or
generalized self-similar profile.  This route is not needed for
\cref{thm:cost-exclusion,cor:absolute-cost-exclusion}.

\begin{definition}[Thick autonomous scale-collapse windows]
A scale-collapse rescaling has thick autonomous windows if there are
\(T>0\), \(c>0\), \(\tau_n\to\infty\), constants
\(a_\infty\in\mathbb R\), \(b_\infty\in\mathbb R^3\), and a constant
\(M_1<\infty\) such that
\[
  \frac1T\int_{\tau_n}^{\tau_n+T}a_-(\tau)M_R(\tau)\,d\tau\ge c,
  \qquad M_R(\tau)\le M_1
\]
for a.e. \(\tau\in[\tau_n,\tau_n+T]\), and
\[
  a(\tau_n+\cdot)\to a_\infty,
  \qquad b(\tau_n+\cdot)\to b_\infty
\]
in a coefficient topology sufficient to pass the coefficient terms in
distributions.  Strong convergence in \(L^1(0,T)\) is sufficient.
\end{definition}

\begin{lemma}[Thick drift gives a negative autonomous scale parameter]\label{lem:negative-parameter}
On thick autonomous scale-collapse windows,
\[
  a_\infty\le -c/M_1<0.
\]
\end{lemma}

\begin{proof}
The upper mass bound and thick weighted drift give
\[
  \frac1T\int_{\tau_n}^{\tau_n+T}a_-(\tau)\,d\tau\ge c/M_1.
\]
After translation and \(L^1(0,T)\) convergence of \(a\) to \(a_\infty\), the
left-hand side converges to \((-a_\infty)_+\).  Therefore
\((-a_\infty)_+\ge c/M_1\), which gives the claim.
\end{proof}

\begin{theorem}[Autonomous reduced-limit route]\label{thm:autonomous-limit}
Consider translated windows
\[
  V_n(y,s)=V(y,\tau_n+s),
  \qquad P_n(y,s)=P(y,\tau_n+s),
  \qquad 0<s<T.
\]
If, on thick autonomous scale-collapse windows, the sequence \((V_n,P_n)\) is
locally compact strongly enough to pass the nonlinear and pressure terms and the
localized core floor passes to a compact subcutoff in the limit, then after
passing to a subsequence \((V_n,P_n)\) converges to a nonzero weak solution
\((U,\Pi)\) of
\begin{equation}\label{eq:autonomous-limit}
  \partial_s U+(U\cdot\nabla)U+\nabla\Pi
  =\nu\Delta U+a_\infty(U+y\cdot\nabla U)+b_\infty\cdot\nabla U,
  \qquad \nabla\cdot U=0.
\end{equation}
\end{theorem}

\begin{proof}
The local \(L^2H^1\) bound and time compactness give, by Aubin--Lions, strong
local convergence in \(L^2(0,T;L^q(B_R))\) for every \(q<6\).  This strong
local convergence, together with weak \(L^2H^1\) convergence, passes the
nonlinear term to \((U\cdot\nabla)U\) in distributions.  The pressure
convergence passes the Poisson reconstruction to \(\Pi\).  The modulation-limit
convergence passes the coefficient terms to
\(a_\infty(U+y\cdot\nabla U)\) and \(b_\infty\cdot\nabla U\).  The localized
\(L^2\) floor passes to the limit on a compact subcutoff, so
\(U\not\equiv0\).
\end{proof}

\begin{theorem}[Stationary-profile rigidity route]\label{thm:stationary-route}
If the autonomous limit in \cref{thm:autonomous-limit} has a nonzero stationary
omega-limit \((W,Q)\) in a class where the corresponding stationary or
generalized self-similar Liouville theorem applies, then the scale-collapse
branch is excluded by that rigidity theorem.
\end{theorem}

\begin{proof}
The stationary omega-limit satisfies
\[
  (W\cdot\nabla)W+\nabla Q
  =\nu\Delta W+a_\infty(W+y\cdot\nabla W)+b_\infty\cdot\nabla W,
  \qquad \nabla\cdot W=0.
\]
It is nonzero by construction.  The applicable Liouville or rigidity theorem
rules out such a nonzero admissible stationary profile, giving a contradiction.
\end{proof}

\begin{remark}[Role of the autonomous route]
The autonomous route is a structural description of a special scale-collapse
subcase.  Infinite scale-drift cost alone does not force thick windows,
coefficient convergence, a stationary omega-limit, or a parameter-specific
Liouville theorem.  Those conditions are not needed for the finite-cost
scale-collapse exclusion, which follows directly from
\cref{thm:cost-exclusion,cor:absolute-cost-exclusion}.
\end{remark}
