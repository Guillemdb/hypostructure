\title[Compact ancient dynamics]{Compact Ancient Navier--Stokes Dynamics and Invariant Measures in Self-similar Variables}

\author{Guillem Duran Ballester}
\date{}

\subjclass[2020]{35Q30, 35B41, 35B44, 37A25, 76D05}
\keywords{Navier--Stokes equations, ancient solutions, invariant measures, self-similar variables, Type I singularities}

\begin{document}

\begin{abstract}
We study compact time-translation dynamics generated by bounded ancient
solutions of the three-dimensional Navier--Stokes equations in backward
self-similar variables.  The compactness used here is stated explicitly:
bounded ancient solutions have compact trajectory hulls in the locally smooth
topology, and uniform \(L^3\)-tightness is stable under passage to the hull.
On every compact invariant hull, Krylov--Bogolyubov averages produce invariant
probability measures.  These measures satisfy the stationary statistical weak
form of the self-similar Navier--Stokes equations.  Passing from this
statistical identity to an exact stationary equation for the barycenter is
equivalent to the vanishing of a precise quadratic covariance.  Thus the paper
separates the invariant-measure construction, which is compactness-theoretic,
from the additional rigidity needed to obtain a single stationary profile.
\end{abstract}

\maketitle

\section{Introduction}

We consider the incompressible Navier--Stokes equations in backward
self-similar variables,
\begin{equation}\label{eq:centered}
   \partial_\tau V+(V\cdot\nabla)V+\nabla P
   =
   \Delta V-\frac12(V+y\cdot\nabla V),
   \qquad \operatorname{div}V=0,
\end{equation}
on \(\mathbb R^3\times\mathbb R\).  The equation is autonomous in the
logarithmic time variable.  If \(V\) is a solution and \(s\in\mathbb R\), then
\[
   V^s(y,\tau)=V(y,\tau+s)
\]
is again a solution.  This paper studies the compact dynamical system obtained
by taking the closure of these time translates in the locally smooth topology.

The preceding minimal-element reduction shows why compact ancient dynamics are
the relevant next object: a nonzero normalized class, once closed under local
limits and time translation, contains a solution of least
\(L^\infty(\mathbb R^3\times\mathbb R)\) size.  The present paper does not
assert that such minimality alone proves recurrence or stationary behavior.
Instead it proves the compact dynamical consequences which are available once
one works on a compact trajectory hull.

There are two points worth separating.  First, the invariant-measure
construction is standard and rigorous on a compact hull.  Time averages of
Dirac masses along an orbit have weak limits, and every such limit is
invariant under time translation.  Second, the nonlinear term remains exact at
the level of statistical solutions: the stationary identity contains the
average of \(W\otimes W\), not an unspecified limiting object.  If one projects
this identity to the barycenter, a quadratic covariance appears.  Eliminating
that covariance is a genuine additional rigidity statement, and it is the
precise bridge from invariant statistical dynamics to a single stationary
self-similar solution.

\section{Ancient solutions and trajectory hulls}

After applying the Leray projection \(\mathbb P\), equation \eqref{eq:centered}
is
\begin{equation}\label{eq:projected}
   \partial_\tau V
   =
   \Delta V-\frac12y\cdot\nabla V-\frac12V
   -\mathbb P\operatorname{div}(V\otimes V).
\end{equation}
Let \(S(a)\), \(a>0\), be the semigroup generated by
\(\Delta-\frac12y\cdot\nabla-\frac12\).

\begin{definition}[Bounded mild ancient solution]\label{def:mild}
A vector field \(V:\mathbb R^3\times\mathbb R\to\mathbb R^3\) is a bounded mild
ancient solution of \eqref{eq:centered} if
\[
   V\in L^\infty(\mathbb R^3\times\mathbb R),\qquad \operatorname{div}V=0
\]
in distributions, and if for every \(\sigma<\tau\)
\begin{equation}\label{eq:mild}
   V(\tau)
   =
   S(\tau-\sigma)V(\sigma)
   -
   \int_\sigma^\tau
   S(\tau-s)\mathbb P\operatorname{div}(V\otimes V)(s)\,ds
\end{equation}
as an identity in the weak-* topology of \(L^\infty(\mathbb R^3)\).
\end{definition}

\begin{definition}[Locally smooth convergence]\label{def:local-smooth}
A sequence \(W_n\) converges locally smoothly to \(W\) on
\(\mathbb R^3\times\mathbb R\) if, for every \(R,T<\infty\) and every integer
\(m\ge0\),
\[
   W_n\to W
   \quad\hbox{in }C^m(B_R\times[-T,T]).
\]
We use this locally smooth topology throughout.  It is the usual Frechet
topology on \(C^\infty_{\mathrm{loc}}(\mathbb R^3\times\mathbb R)\), and in
particular it is metrizable.
\end{definition}

\begin{lemma}[Local smoothing]\label{lem:smoothing}
Let \(V\) be a bounded mild solution of \eqref{eq:centered} on
\(\mathbb R^3\times(\tau_0,\tau_1)\).  For every compact cylinder
\[
   Q\Subset \mathbb R^3\times(\tau_0,\tau_1)
\]
and every integer \(m\ge0\),
\[
   \|V\|_{C^m(Q)}
   \le
   C\bigl(m,Q,\tau_0,\tau_1,
   \|V\|_{L^\infty(\mathbb R^3\times(\tau_0,\tau_1))}\bigr).
\]
The same estimate is uniform for families with a common \(L^\infty\) bound.
\end{lemma}

\begin{proof}
Fix a compact interval \(J\Subset(\tau_0,\tau_1)\) containing the time
projection of \(Q\), and set \(t=-e^{-\tau}\).  On the corresponding compact
negative time interval define
\[
   U(x,t)=(-t)^{-1/2}V(x/\sqrt{-t},-\log(-t)).
\]
Then \(U\) is a bounded mild solution of the usual Navier--Stokes equations on
that interval, with a bound depending only on \(J\) and the displayed
\(L^\infty\) norm of \(V\).  Interior smoothing for bounded mild
Navier--Stokes solutions, as in Kato \cite{Kato1984} and Giga--Miyakawa
\cite{GigaMiyakawa1985}, gives bounds for all derivatives of \(U\) on compact
subcylinders.  Returning to \((y,\tau)\) gives the asserted estimate.
\end{proof}

\begin{proposition}[Compactness of bounded families]\label{prop:bounded-compactness}
Let \(V_n\) be bounded mild ancient solutions of \eqref{eq:centered} satisfying
\[
   \sup_n\|V_n\|_{L^\infty(\mathbb R^3\times\mathbb R)}<\infty .
\]
Then a subsequence converges locally smoothly to a bounded mild ancient
solution of \eqref{eq:centered}.
\end{proposition}

\begin{proof}
\Cref{lem:smoothing} gives uniform \(C^m\) bounds on every compact cylinder.
Arzela--Ascoli and a diagonal argument give local smooth convergence along a
subsequence.  Passing to the limit in the mild formulation is standard:
testing \eqref{eq:mild} against \(C_c^\infty\) functions, the adjoint Stokes
kernel has the integrable bound
\[
   C e^{-(\tau-s)/2}(1-e^{-(\tau-s)})^{-1/2}
\]
near \(s=\tau\).  Local uniform convergence and the common \(L^\infty\) bound
then give dominated convergence in the Duhamel term.  For the noncompact
spatial tails of the adjoint kernel, one first approximates the \(L^1\) kernel
by compactly supported functions; the common \(L^\infty\) bound controls the
tail uniformly.  Density of \(C_c^\infty(\mathbb R^3)\) in
\(L^1(\mathbb R^3)\) yields the weak-* identity \eqref{eq:mild} for the limit.
\end{proof}

\begin{theorem}[Compact trajectory hull]\label{thm:compact-hull}
Let \(V\) be a bounded mild ancient solution of \eqref{eq:centered}.  Let \(K\)
be the closure, in the locally smooth topology, of the family
\[
   \{V^s:\ s\in\mathbb R\}.
\]
Then \(K\) is compact.  Every \(W\in K\) is a bounded mild ancient solution of
\eqref{eq:centered}, with
\[
   \|W\|_{L^\infty(\mathbb R^3\times\mathbb R)}
   \le
   \|V\|_{L^\infty(\mathbb R^3\times\mathbb R)}.
\]
Moreover the maps
\[
   \Theta_t W=W^t,\qquad t\in\mathbb R,
\]
define a continuous group of transformations of \(K\).

If, in addition, \(V\) satisfies
\begin{equation}\label{eq:L3-tight}
   \sup_{\tau\in\mathbb R}\|V(\tau)\|_{L^3(\mathbb R^3)}\le L
\end{equation}
and there is \(\omega:[1,\infty)\to[0,\infty)\), with
\(\omega(R)\to0\) as \(R\to\infty\), such that
\begin{equation}\label{eq:tail}
   \sup_{\tau\in\mathbb R}
   \int_{|y|>R}|V(y,\tau)|^3\,dy\le\omega(R)
   \qquad R\ge1,
\end{equation}
then every \(W\in K\) satisfies the corresponding estimates
\[
   \sup_{\tau\in\mathbb R}\|W(\tau)\|_{L^3(\mathbb R^3)}\le L
\]
and
\[
   \sup_{\tau\in\mathbb R}
   \int_{|y|>R}|W(y,\tau)|^3\,dy\le\omega(R),
   \qquad R\ge1 .
\]
\end{theorem}

\begin{proof}
Every sequence of translates \(V^{s_n}\) satisfies the uniform
\(L^\infty\) bound of \(V\).  By \cref{prop:bounded-compactness}, a subsequence
converges locally smoothly to a bounded mild ancient solution.  Thus every
element of \(K\) is a bounded mild ancient solution with the displayed
\(L^\infty\) bound.

Now let \(W_n\in K\).  The preceding paragraph gives a common \(L^\infty\)
bound for the sequence.  By \cref{prop:bounded-compactness}, a subsequence
converges locally smoothly to a bounded mild ancient solution
\(\widetilde W\).  Since \(K\) is closed by definition, \(\widetilde W\in K\).
Thus \(K\) is sequentially compact.  The locally smooth topology is
metrizable, so \(K\) is compact.

The \(L^\infty\) bound passes to the limit pointwise.  The identity
\(\Theta_t\Theta_s=\Theta_{t+s}\) is immediate from the definition of time
translation, and continuity follows directly from locally smooth convergence
on compact time intervals.

It remains to check the uniform \(L^3\) assertions.  Let \(W_n=V^{s_n}\to W\)
locally smoothly.  For fixed \(\tau\) and \(R<\infty\),
\[
   \int_{B_R}|W(y,\tau)|^3\,dy
   =
   \lim_{n\to\infty}\int_{B_R}|W_n(y,\tau)|^3\,dy
   \le L^3 .
\]
Letting \(R\to\infty\) gives \eqref{eq:L3-tight} for \(W\).  Similarly, for
fixed \(1\le R<R'<\infty\),
\[
   \int_{R<|y|<R'}|W(y,\tau)|^3\,dy
   =
   \lim_{n\to\infty}\int_{R<|y|<R'}|W_n(y,\tau)|^3\,dy
   \le\omega(R).
\]
Letting \(R'\to\infty\) and then taking the supremum over \(\tau\) gives
\eqref{eq:tail} for \(W\).
\end{proof}

\section{Compact invariant sets}

\begin{definition}[Compact invariant set]\label{def:compact-invariant}
A nonempty compact subset \(K\), with respect to the locally smooth topology,
of bounded mild ancient solutions is invariant if \(\Theta_tK=K\) for every
\(t\in\mathbb R\).  It is minimal if it has no proper nonempty compact
invariant subset.
\end{definition}

\begin{proposition}[Minimal compact subsets]\label{prop:minimal-subset}
Every nonempty compact invariant set contains a minimal compact invariant
subset.
\end{proposition}

\begin{proof}
Order the nonempty compact invariant subsets by inclusion.  If
\(\{K_\alpha\}\) is a decreasing chain, then \(\bigcap_\alpha K_\alpha\) is
nonempty by compactness and compact by closedness.  It is invariant because
each \(\Theta_t\) is a homeomorphism and
\[
   \Theta_t\left(\bigcap_\alpha K_\alpha\right)
   =
   \bigcap_\alpha \Theta_tK_\alpha
   =
   \bigcap_\alpha K_\alpha .
\]
Zorn's lemma gives a minimal element.
\end{proof}

\begin{lemma}[The zero trajectory]\label{lem:zero-trajectory}
Let \(K\) be a minimal compact invariant set.  If the zero solution belongs to
\(K\), then \(K\) consists only of the zero solution.
\end{lemma}

\begin{proof}
The singleton containing the zero solution is compact and invariant.  By
minimality it must equal \(K\).
\end{proof}

\section{Invariant measures}

\begin{theorem}[Krylov--Bogolyubov averages]\label{thm:KB}
Let \(K\) be a compact invariant set and let \(W_0\in K\).  For \(T>0\), define
the probability measure
\[
   \mu_T=\frac1T\int_0^T\delta_{\Theta_sW_0}\,ds
\]
on \(K\).  If \(T_n\to\infty\), then a subsequence of \(\mu_{T_n}\) converges
weakly to a Borel probability measure \(\mu\) on \(K\).  Every such limit is
invariant:
\[
   \int_K F(\Theta_tW)\,d\mu(W)=\int_K F(W)\,d\mu(W)
\]
for every \(t\in\mathbb R\) and every continuous function \(F:K\to\mathbb R\).
If \(K\) is minimal, then the support of every invariant Borel probability
measure on \(K\) is equal to \(K\).
\end{theorem}

\begin{proof}
Since \(K\) is compact and metrizable, the set of Borel probability measures
on \(K\) is weakly compact.  Hence a subsequential weak limit exists.

Let \(F\) be continuous and fix \(t>0\).  Then
\[
   \int_K F(\Theta_tW)\,d\mu_T(W)-\int_K F(W)\,d\mu_T(W)
   =
   \frac1T\int_T^{T+t}F(\Theta_sW_0)\,ds
   -
   \frac1T\int_0^tF(\Theta_sW_0)\,ds .
\]
The absolute value of the right-hand side is at most
\[
   \frac{2t}{T}\|F\|_{L^\infty(K)}.
\]
Letting \(T=T_n\to\infty\) along the convergent subsequence gives invariance
for \(t>0\).  If \(t<0\), apply the positive-time result with \(-t\) to the
continuous function \(F\circ\Theta_t\).  Since
\(\Theta_{-t}\Theta_t\) is the identity, this gives the asserted identity for
negative \(t\) as well.

If \(K\) is minimal and \(\mu\) is invariant, then the support of \(\mu\) is
nonempty and compact.  It is also invariant.  Indeed, if
\(W\in\operatorname{supp}\mu\) and \(U\) is an open neighborhood of
\(\Theta_tW\), then \(\Theta_{-t}U\) is an open neighborhood of \(W\), hence
\[
   \mu(U)=\mu(\Theta_{-t}U)>0 .
\]
Thus \(\Theta_tW\in\operatorname{supp}\mu\).  Applying the same argument to
\(-t\) gives equality.  The support is therefore a nonempty compact invariant
subset of \(K\), and minimality forces it to be \(K\).
\end{proof}

\begin{corollary}[Nonzero support]\label{cor:nonzero-support}
Let \(K\) be a minimal compact invariant set containing at least one nonzero
trajectory.  Then the zero trajectory is not in \(K\), and every invariant
Borel probability measure on \(K\) has support equal to \(K\).  In particular,
no such measure is supported at the zero solution.
\end{corollary}

\begin{proof}
The first assertion follows from \cref{lem:zero-trajectory}.  The support
statement is the last assertion of \cref{thm:KB}.
\end{proof}

\section{The stationary statistical identity}

The next theorem records the exact weak identity satisfied by invariant
measures.  The test functions are divergence free, so the pressure does not
appear.

\begin{theorem}[Stationary statistical form]\label{thm:statistical-identity}
Let \(K\) be a compact invariant set of bounded mild ancient solutions of
\eqref{eq:centered}, and let \(\mu\) be an invariant Borel probability measure
on \(K\).  Then, for every \(\phi\in C_c^\infty(\mathbb R^3;\mathbb R^3)\) with
\(\operatorname{div}\phi=0\),
\begin{equation}\label{eq:statistical-identity}
   \begin{aligned}
   0
   =
   \int_K
   \bigg[
      &\int_{\mathbb R^3}
      W(y,0)\cdot
      \left(\Delta\phi(y)+\phi(y)+\frac12 y\cdot\nabla\phi(y)\right)\,dy \\
      &+
      \int_{\mathbb R^3}
      W_i(y,0)W_j(y,0)\partial_j\phi_i(y)\,dy
   \bigg]d\mu(W).
   \end{aligned}
\end{equation}
\end{theorem}

\begin{proof}
For such a test function define
\[
   F(W)=\int_{\mathbb R^3}W(y,0)\cdot\phi(y)\,dy .
\]
Invariance gives
\[
   \int_K F(\Theta_hW)\,d\mu(W)=\int_K F(W)\,d\mu(W)
   \qquad h\in\mathbb R .
\]
Therefore
\[
   \int_K \frac{F(\Theta_hW)-F(W)}{h}\,d\mu(W)=0 .
\]
Because \(K\) is compact in the locally smooth topology, all trajectories in
\(K\) have a common \(C^1\) bound on a fixed compact cylinder containing the
support of \(\phi\) and times \(|\tau|\le1\).  Thus the difference quotients are
uniformly bounded for \(|h|\le1\), and dominated convergence gives
\[
   \int_K\int_{\mathbb R^3}\partial_\tau W(y,0)\cdot\phi(y)\,dy\,d\mu(W)=0 .
\]
Using \eqref{eq:centered}, integrating by parts, and using
\(\operatorname{div}W=\operatorname{div}\phi=0\), we obtain for each \(W\)
\[
   \begin{aligned}
   \int_{\mathbb R^3}\partial_\tau W(y,0)\cdot\phi(y)\,dy
   =
   &\int_{\mathbb R^3}
   W(y,0)\cdot
   \left(\Delta\phi(y)+\phi(y)+\frac12 y\cdot\nabla\phi(y)\right)\,dy\\
   &+
   \int_{\mathbb R^3}
   W_i(y,0)W_j(y,0)\partial_j\phi_i(y)\,dy .
   \end{aligned}
\]
Substitution gives \eqref{eq:statistical-identity}.
\end{proof}

\begin{theorem}[Barycenter and covariance]\label{thm:barycenter}
Let \(K\) and \(\mu\) be as in \cref{thm:statistical-identity}.  Define the
barycenter
\[
   \overline W(y)=\int_K W(y,0)\,d\mu(W)
\]
and the quadratic covariance
\[
   C_{ij}(y)=
   \int_K W_i(y,0)W_j(y,0)\,d\mu(W)
   -
   \overline W_i(y)\overline W_j(y).
\]
Then \(\overline W\) is smooth and divergence free.  Moreover, for every
divergence-free
\(\phi\in C_c^\infty(\mathbb R^3;\mathbb R^3)\),
\begin{equation}\label{eq:barycenter-equation}
   \int_{\mathbb R^3}
   \overline W\cdot
   \left(\Delta\phi+\phi+\frac12 y\cdot\nabla\phi\right)\,dy
   +
   \int_{\mathbb R^3}
   \overline W_i\overline W_j\partial_j\phi_i\,dy
   +
   \int_{\mathbb R^3}
   C_{ij}\partial_j\phi_i\,dy
   =0 .
\end{equation}
Consequently, \(\overline W\) is a weak stationary solution of
\eqref{eq:centered}, in the pressure-free formulation against divergence-free
test fields, if and only if
\begin{equation}\label{eq:zero-covariance}
   \int_{\mathbb R^3}C_{ij}(y)\partial_j\phi_i(y)\,dy=0
\end{equation}
for every divergence-free
\(\phi\in C_c^\infty(\mathbb R^3;\mathbb R^3)\).
\end{theorem}

\begin{proof}
The compactness of \(K\) in the locally smooth topology justifies
differentiating the barycenter under the integral sign on compact sets.  Since
every trajectory in \(K\) is divergence free, \(\overline W\) is divergence
free.  Apply \cref{thm:statistical-identity} and split
\[
   \int_K W_iW_j\,d\mu
   =
   \overline W_i\overline W_j+C_{ij}.
\]
This gives \eqref{eq:barycenter-equation}.  The weak stationary form of
\eqref{eq:centered} for \(\overline W\), tested against divergence-free
\(\phi\), is exactly \eqref{eq:barycenter-equation} without the last term.
Thus it is equivalent to \eqref{eq:zero-covariance}.
\end{proof}

\begin{corollary}[Stationary support gives a stationary profile]\label{cor:stationary-support}
Let \(K\) be a compact invariant set.  Assume that there is \(L<\infty\) such
that every \(W\in K\) satisfies
\[
   \sup_{\tau\in\mathbb R}\|W(\tau)\|_{L^3(\mathbb R^3)}\le L .
\]
Let \(\mu\) be an invariant probability measure on \(K\).  If \(\mu\) gives
positive mass to the set of stationary trajectories, then \(K\) contains a
stationary solution \(W\) with \(W\in L^3(\mathbb R^3)\).  If, in addition,
the stationary trajectory is nonzero, this contradicts the stationary
self-similar theorem of Necas--Ruzicka--Sverak.
\end{corollary}

\begin{proof}
The set of stationary trajectories is closed in the locally smooth topology.
If it has positive \(\mu\)-measure, it is nonempty.  Any element of it has the
form \(W(y,\tau)=W(y,0)\) and solves the stationary self-similar equation
\[
   (W\cdot\nabla)W+\nabla P
   =
   \Delta W-\frac12(W+y\cdot\nabla W),
   \qquad \operatorname{div}W=0 .
\]
The uniform \(L^3\) bound gives \(W\in L^3(\mathbb R^3)\).  The theorem of
Necas--Ruzicka--Sverak \cite{NRS1996} then forces every such stationary
solution to vanish.
\end{proof}

\section{Conclusion}

Bounded ancient solutions generate compact trajectory hulls in the locally
smooth topology, and uniform \(L^3\)-tightness is stable on those hulls.
Compact invariant subsets support invariant probability measures obtained by
time averaging.  These measures satisfy an exact stationary statistical weak
form of the self-similar Navier--Stokes equations.  The obstruction to turning
the statistical identity into a single stationary solution is precisely the
quadratic covariance in \cref{thm:barycenter}.  A later rigidity argument must
show, under additional hypotheses, that compact nonzero ancient dynamics place
mass on stationary trajectories or otherwise make this covariance vanish.
