\title[Local Caccioppoli and Rough-Core Exclusion]{Local Caccioppoli Estimate and Rough-Core Exclusion}
\author{}
\date{}

\begin{document}

\begin{abstract}
We present a fully explicit local Caccioppoli estimate for renormalized Navier--Stokes fields. All terms produced by the localized energy inequality are estimated separately, and all absorptions are written explicitly. The argument uses standard PDE notation and the usual integrability classes of suitable weak solutions.

In particular, we derive local windowed $H^1$ control on compact space--time cylinders from local compact Caffarelli--Kohn--Nirenberg quantities. The result gives the rough-core alternative as the negation of local compact-CKN control.

This paper is written in the classical framework of suitable weak solutions (Leray~\cite{Leray1934}, Caffarelli--Kohn--Nirenberg~\cite{CaffarelliKohnNirenberg1982}, Scheffer~\cite{Scheffer1976}, Lin~\cite{Lin1998}).
\end{abstract}

\section{Localized equation and energy inequality}

Let $(V,P,a,b)$ be defined on the compact renormalized cylinder
\[
Q_{2R,I}=B_{2R}\times I,
\]
where $I\subset\mathbb R$ is a bounded interval and $B_{2R}\subset\mathbb R^3$ is the ball of radius $2R$. Assume
\begin{equation}\label{eq:renorm-ns}
\partial_\tau V+(V\cdot\nabla)V+\nabla P-\nu\Delta V
= a(\tau)V+a(\tau)(y\cdot\nabla V)+b(\tau)\cdot\nabla V,
\qquad
\nabla\cdot V=0,
\end{equation}
with viscosity constant $\nu>0$.

Throughout, for a space--time set $E\subset\mathbb R^3\times\mathbb R$, let
\[
\int_E f:=\int_E f(y,\tau)\,dy\,d\tau,
\]
and for a ball $B\subset\mathbb R^3$, denote the spatial mean of pressure by
\[
(P)_B(\tau)=\frac1{|B|}\int_B P(y,\tau)\,dy.
\]

A suitable weak solution on $Q_{2R,I}$ satisfies the local energy inequality: for every nonnegative
$\phi\in C_c^\infty(Q_{2R,I})$ and every $\tau_1<\tau_2$ in $I$,
\begin{equation}\label{eq:local-energy}
\begin{aligned}
\int_{B_{2R}}\frac{|V(\cdot,\tau_2)|^2}{2}\phi(\cdot,\tau_2)
-\int_{B_{2R}}\frac{|V(\cdot,\tau_1)|^2}{2}\phi(\cdot,\tau_1)
+\nu\int_{\tau_1}^{\tau_2}\!\int_{B_{2R}}\phi\,|\nabla V|^2
&\le
\int_{\tau_1}^{\tau_2}\!\int_{B_{2R}}\frac{|V|^2}{2}(\partial_\tau\phi+\Delta\phi)
+\int_{\tau_1}^{\tau_2}\!\int_{B_{2R}}\bigl(\tfrac{|V|^2}{2}V+PV\bigr)\cdot\nabla\phi
\\
&\quad+\int_{\tau_1}^{\tau_2}\!\int_{B_{2R}}\bigl(aV+a(y\cdot\nabla V)+b\cdot\nabla V\bigr)\cdot V\phi.
\end{aligned}
\end{equation}

\section{Compact-cylinder Caccioppoli estimate}

\begin{theorem}[Compact-cylinder Caccioppoli estimate]\label{thm:caccioppoli}
Assume that
\[
V\in L_t^\infty L_x^2(Q_{2R,I})\cap L_t^2 H_x^1(Q_{2R,I}),\qquad
P\in L_t^{3/2}L_x^{3/2}(Q_{2R,I}),
\]
and
\[
a,b\in L^\infty(I),\qquad
\int_{Q_{2R,I}}|V|^3<\infty,
\qquad
\int_{Q_{2R,I}}|P-(P)_{B_{2R}}(\tau)|^{3/2}<\infty.
\]
Let $I'\Subset I$ be compact. Then there is a constant
\[
C=C\bigl(R,\nu,\|a\|_{L^\infty(I)},\|b\|_{L^\infty(I)},
\operatorname{dist}(I',\partial I)\bigr)
\]
such that
\[
\int_{I'}\int_{B_R}|\nabla V|^2
\le C\Bigl(1+
\int_I\!\int_{B_{2R}}|V|^3+
\int_I\!\int_{B_{2R}}|P-(P)_{B_{2R}}(\tau)|^{3/2}
\Bigr).
\]
\end{theorem}

\begin{proof}
\textbf{Step 1: choice of cutoff and time truncation.}
Choose
\[
\phi(y,\tau)=\eta(\tau)\,\zeta(y)^2,
\]
where
\[
\zeta\in C_c^\infty(B_{2R}),\quad 0\le\zeta\le1,\quad
\zeta\equiv1 \text{ on }B_R,
\]
\[
\eta\in C_c^\infty(\mathbb R),\quad 0\le\eta\le1,
\quad \eta\equiv1\text{ on }I',
\]
with compact support in $I$. Because $I'\Subset I$, there exist times $\tau_1<\tau_2$ in $I$ such that
\[
\eta(\tau_1)=\eta(\tau_2)=0,
\quad
\eta\equiv1\text{ on }I'\subset(\tau_1,\tau_2).
\]

Insert this cutoff in \eqref{eq:local-energy}. Since $\phi(\cdot,\tau_1)=\phi(\cdot,\tau_2)=0$, the terminal kinetic terms vanish, and we obtain
\[
\nu\int_{\tau_1}^{\tau_2}\!\int_{B_{2R}}\phi\,|\nabla V|^2
\le |R_\tau|+|R_\Delta|+|R_{\mathrm{conv}}|+|R_{\mathrm{pres}}|+|R_{\mathrm{mod}}|,
\]
where all integrals are over $Q_{2R,I}$ and
\[
\begin{aligned}
R_\tau&:=\int \frac{|V|^2}{2}\,\partial_\tau\phi,
\quad R_\Delta:=\int \frac{|V|^2}{2}\,\Delta\phi,\\
R_{\mathrm{conv}}&:=\int \bigl(\tfrac{|V|^2}{2}V\bigr)\cdot\nabla\phi,
\quad
R_{\mathrm{pres}}:=\int \bigl(P-(P)_{B_{2R}}(\tau)\bigr)V\cdot\nabla\phi,
\quad
R_{\mathrm{mod}}:=\int \bigl(aV+a(y\cdot\nabla V)+b\cdot\nabla V\bigr)\cdot V\phi.
\end{aligned}
\]
Define
\[
A:=\int_{Q_{2R,I}}|V|^2,
\quad
B:=\int_{Q_{2R,I}}|V|^3,
\quad
D:=\int_{Q_{2R,I}}|P-(P)_{B_{2R}}(\tau)|^{3/2}.
\]

\textbf{Step 2: time--derivative and Laplacian terms.}
Since $\phi(y,\tau)=\eta(\tau)\zeta(y)^2$,
\[
\partial_\tau\phi=\eta'(\tau)\zeta^2,
\qquad
\Delta\phi=\eta\Delta(\zeta^2).
\]
Hence
\[
|R_\tau|
\le\frac12\|\eta'\|_{L^\infty}\,\|\zeta\|_{L^\infty}^2\,A,
\quad
|R_\Delta|
\le\frac12\|\eta\|_{L^\infty}\,\|\Delta(\zeta^2)\|_{L^\infty}\,A.
\]
Therefore
\[
|R_\tau|+|R_\Delta|\le C_0A.
\]
By Hölder in space and time,
\[
A\le|I|^{1/3}|B_{2R}|^{2/3}B^{2/3}\le C_{R,I}(1+B),
\]
using $s^{2/3}\le1+s$ for $s\ge0$. So
\[
|R_\tau|+|R_\Delta|\le C_0\bigl(1+B\bigr).
\]

\textbf{Step 3: convection term.}
Using $\nabla\phi=\eta\nabla(\zeta^2)$ and the support property of $\nabla\zeta$,
\[
|R_{\mathrm{conv}}|
\le \|\nabla\phi\|_{L^\infty}\int_{Q_{2R,I}}|V|^3
\le C(R)\,B.
\]

\textbf{Step 4: pressure term.}
The identity
\[
\int (P)_{B_{2R}}(\tau)\,V\cdot\nabla\phi=0
\]
holds because $\int_{B_{2R}}\nabla\phi=0$. By Hölder,
\[
|R_{\mathrm{pres}}|
\le \|\nabla\phi\|_{L^\infty}\|P-(P)_{B_{2R}}(\tau)
\|_{L_t^{3/2}L_x^{3/2}(Q_{2R,I})}\|V\|_{L_t^3L_x^3(Q_{2R,I})}.
\]
Since
\[
\|P-(P)_{B_{2R}}\|_{L_t^{3/2}L_x^{3/2}}=D^{2/3},
\qquad
\|V\|_{L_t^3L_x^3}=B^{1/3},
\]
Young's inequality with exponents $3/2$ and $3$ gives
\[
|R_{\mathrm{pres}}|\le \frac12 D + C(R) B.
\]

\textbf{Step 5: modulation terms.}
Write
\[
R_{\mathrm{mod}}=R_a+R_{ay}+R_b,
\]
with
\[
R_a:=\int a\phi|V|^2,
\quad
R_{ay}:=\int a\phi(y\cdot\nabla V)\cdot V,
\quad
R_b:=\int \phi(b\cdot\nabla V)\cdot V.
\]
For $R_a$,
\[
|R_a|\le \|a\|_{L^\infty(I)}A
\le C_a\bigl(1+B\bigr),
\]
using Step~2.\
For $R_{ay}$, since $|y|\le2R$ on $\operatorname{supp}\phi$,
\[
|R_{ay}|\le\|a\|_{L^\infty(I)}
\left(\int \phi|\nabla V|^2\right)^{1/2}
\left(\int \phi|y|^2|V|^2\right)^{1/2}
\le\|a\|_{L^\infty(I)}
\left(\int \phi|\nabla V|^2\right)^{1/2}
\left(\int\phi|V|^2\right)^{1/2}(2R).
\]
Applying Young's inequality with parameter $\nu/8$ and Step~2 yields
\[
|R_{ay}|
\le \frac{\nu}{8}\int\phi|\nabla V|^2+
C\frac{R^2\|a\|_{L^\infty(I)}^2}{\nu}\,(1+B).
\]
For $R_b$,
\[
|R_b|\le\|b\|_{L^\infty(I)}
\left(\int\phi|\nabla V|^2\right)^{1/2}
\left(\int\phi|V|^2\right)^{1/2}
\le \frac{\nu}{8}\int\phi|\nabla V|^2+
C\frac{\|b\|_{L^\infty(I)}^2}{\nu}\,(1+B).
\]

\textbf{Step 6: absorption and conclusion.}
Summing all estimates gives
\[
\nu\int\phi|\nabla V|^2
\le C_1(1+B)+C_2B+\tfrac12D+
\frac{\nu}{8}\int\phi|\nabla V|^2+\frac{\nu}{8}\int\phi|\nabla V|^2.
\]
Hence
\[
\frac\nu2\int\phi|\nabla V|^2
\le C_3\bigl(1+B\bigr)+C_4D.
\]
Since $\phi\ge1$ on $B_R\times I'$, this gives
\[
\int_{I'}\int_{B_R}|\nabla V|^2
\le C\Bigl(1+
\int_I\!\int_{B_{2R}}|V|^3+
\int_I\!\int_{B_{2R}}|P-(P)_{B_{2R}}(\tau)|^{3/2}
\Bigr),
\]
with $C$ depending only on the quantities specified in the theorem.
\end{proof}

\begin{definition}[Local Caffarelli--Kohn--Nirenberg control on compact cylinders]
For $r>0$ and compact $J\subset I$, define
\[
\mathcal C_J(r):=\int_J\!\int_{B_r}|V|^3,
\qquad
\mathcal D_J(r):=\int_J\!\int_{B_r}|P-(P)_{B_r}(\tau)|^{3/2}.
\]
We say that local Caffarelli--Kohn--Nirenberg control holds on $J\times B_r$ if both
$\mathcal C_J(r)$ and $\mathcal D_J(r)$ are finite.
\end{definition}

\begin{corollary}[Rough-core alternative]
\label{cor:rough-core}
Assume $a,b\in L^\infty(I)$ and let $R>0$, $I'\Subset I$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item If local Caffarelli--Kohn--Nirenberg control holds on $I\times B_{2R}$,
then for every compact $I'\Subset I$ there exists
\[
C=C(R,I',\nu,\|a\|_{L^\infty(I)},\|b\|_{L^\infty(I)},\mathcal C_I(2R),\mathcal D_I(2R))
\]
such that
\[
\sup_{\tau_1<\tau_2\in I'}\int_{\tau_1}^{\tau_2}\int_{B_R}|\nabla V|^2
\le C.
\]
\item If the local windowed gradient bound fails on every inner ball $B_{R'}\subset B_R$
around a candidate singular location, then local Caffarelli--Kohn--Nirenberg control fails on
some compact inner cylinder.
\[
\text{rough-core windowed }H^1\text{ control fails}
\Longrightarrow
\exists\ R'\in(0,R],\;\exists\ J\Subset I:
\bigl(\mathcal C_J(R')=+\infty\text{ or }\mathcal D_J(R')=+\infty\bigr).
\]
\end{enumerate}
\end{corollary}

\begin{proof}
\textbf{(i)} If local CKN control holds on $I\times B_{2R}$, then
$\mathcal C_I(2R)$ and $\mathcal D_I(2R)$ are finite. Apply Theorem~\ref{thm:caccioppoli}
with this data. The conclusion is exactly the stated bound.

\textbf{(ii)} We prove the contrapositive. Suppose there exists a compact cylinder
$J\times B_{R'}$ with $R'\in(0,R]$ such that local CKN control holds, i.e.
$\mathcal C_J(R')<\infty$ and $\mathcal D_J(R')<\infty$. Theorem~\ref{thm:caccioppoli}
applied on radius $R'$ and interval $J$ gives a finite constant $C$ with
\[
\sup_{\sigma_1<\sigma_2\in J}\int_{\sigma_1}^{\sigma_2}\int_{B_{R'}}|\nabla V|^2\le C,
\]
so the local windowed gradient bound holds on that cylinder. Therefore, if the local windowed gradient bound
fails on every inner ball and every compact time window, local CKN control must fail on at least
one compact cylinder.
\end{proof}
